import json,os,pathlib,shlex,subprocess,time
old='1b3b298f-051b-42ea-a53b-429c16f4e114'
script='import json,os,pathlib;print(json.dumps(dict(boot_id=pathlib.Path("/proc/sys/kernel/random/boot_id").read_text().strip(),release=os.uname().release,taint=pathlib.Path("/proc/sys/kernel/tainted").read_text().strip())))'

journal=pathlib.Path('/tmp/kdlock-kernel4-db7-return-reconnect-attempts.jsonl').open('x')
while time.monotonic()-start<240:
 row={'unix':time.time(),'argv':argv}
 try:
  r=subprocess.run(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,timeout=min(20,max(1,240-(time.monotonic()-start))))
  row.update(returncode=r.returncode,stdout=r.stdout,stderr=r.stderr)

   d=json.loads(r.stdout);row['identity']=d
   success=d['boot_id']!=old and d['release']=='7.0.0-38-kdlock1' and d['taint']=='0'
 except (subprocess.TimeoutExpired,ValueError,KeyError) as e:
  row['error']=repr(e)
 rows.append(row);journal.write(json.dumps(row)+'\n');journal.flush();os.fsync(journal.fileno())
 if success:break

 if remaining>0:time.sleep(min(10,remaining))
journal.close()
result={'status':'NORMAL_KERNEL_OBSERVED' if success else 'NO_VERIFIED_NEW_BOOT','elapsed_seconds':time.monotonic()-start,'old_boot_id':old,'attempts':rows}
with pathlib.Path('/tmp/kdlock-kernel4-db7-return-reconnect-20261111-01.json').open('x') as f:json.dump(result,f,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
print(json.dumps(result,indent=2))
raise SystemExit(0 if success else 1)
