def sha(value):
    return hashlib.sha256(value).hexdigest()


def write(path, value):
    with path.open('xb') as stream:
        stream.write(value)
        stream.flush()
        os.fsync(stream.fileno())


def document(path, value):
    write(path, (json.dumps(value, indent=2, sort_keys=True) + '\n').encode())

def remote(code, seconds, maximum):
    argv = SSH + ['timeout '+str(seconds)+'s python3 -']
    start = time.time()
    result = subprocess.run(argv, input=code.encode(), stdout=subprocess.PIPE,
                            stderr=subprocess.PIPE, timeout=seconds+10)
    record = {'argv': argv, 'script_sha256': sha(code.encode()), 'returncode': result.returncode,
              'started_unix': start, 'finished_unix': time.time(),
              'stdout_bytes': len(result.stdout), 'stdout_sha256': sha(result.stdout),
              'stderr': result.stderr.decode(errors='replace')}
    if result.returncode or len(result.stdout)>maximum or len(result.stderr)>16*1024**2:
        raise RuntimeError(record)
    return result.stdout, record

        with tarfile.open(fileobj=io.BytesIO(payload)) as archive:
            for member in archive:
                parts=PurePosixPath(member.name).parts
                if not member.isfile() or member.name.startswith('/') or '..' in parts or member.name not in index['files'] or member.name in contents:
                    raise RuntimeError('unexpected copy member')
                row=index['files'][member.name]
                if member.size!=row['bytes'] or member.size>MAX_FILE:
                    raise RuntimeError('copy member size differs')
                value=archive.extractfile(member).read(MAX_FILE+1)
                if len(value)!=row['bytes'] or sha(value)!=row['sha256']:
                    raise RuntimeError('copied file hash differs')
                contents[member.name]=value
