def save():
    (receipts / 'result.json').write_text(json.dumps(record, indent=2, sort_keys=True) + '\n')

def phase(name):
    record['phase'] = name

def fixture_status(name):
    os.lseek(control, 0, os.SEEK_SET)
    raw = os.read(control, 16384)
    (receipts / name).write_bytes(raw)
    values = dict(item.split('=', 1) for item in raw.decode('ascii').split())
    return values

def send(operation, sequence, fixed=(), rows=(), blobs=(), bounds=()):
    payload = encode(operation, fixed, rows, blobs, bounds, limits)
    name = f'{sequence:02d}-{operation.name.lower()}'
    (receipts / (name + '-request.bin')).write_bytes(payload)
    phase(name + '_DISPATCHING')
    result = transport.exchange(operation, payload)
    (receipts / (name + '-reply.bin')).write_bytes(result.payload)
    item = {'operation': int(operation), 'sequence': result.sequence, 'status': result.status,
            'request_length': len(payload), 'reply_length': len(result.payload)}
    record['exchanges'].append(item)
    phase(name + '_REPLIED')
    if result.sequence != sequence or result.status != 0:
        raise RuntimeError('unexpected actual sequence/status: ' + repr(item))
    return result.payload

def exact(operation, sequence, fixed):
    raw = send(operation, sequence, fixed)
    frame = decode(raw, operation, len(fixed), 0, (), limits)
    if frame.fixed != fixed:
        raise RuntimeError('actual reply identity differs: ' + repr(frame.fixed))
    return raw
